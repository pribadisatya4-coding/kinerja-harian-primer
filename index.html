Pengaturan Database Supabase

Hubungkan aplikasi dengan Supabase agar data pegawai dan laporan tersimpan permanen lintas perangkat.

Skrip SQL Setup Tabel (Jalankan di SQL Editor Supabase) 
-- Tabel Pegawai
CREATE TABLE IF NOT EXISTS employees (
  id TEXT PRIMARY KEY,
  name TEXT NOT NULL,
  nip TEXT NOT NULL,
  role TEXT NOT NULL,
  division TEXT NOT NULL,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Tabel Laporan Kinerja Harian
CREATE TABLE IF NOT EXISTS reports (
  id TEXT PRIMARY KEY,
  employee_id TEXT REFERENCES employees(id) ON DELETE CASCADE,
  report_date DATE NOT NULL,
  start_time TEXT NOT NULL,
  end_time TEXT NOT NULL,
  duration_minutes INTEGER NOT NULL,
  title TEXT NOT NULL,
  output TEXT NOT NULL,
  index_val NUMERIC(4,2) DEFAULT 1.0,
  points NUMERIC(6,2) DEFAULT 0,
  status TEXT DEFAULT 'Selesai',
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Aktifkan RLS & Kebijakan Akses Publik
ALTER TABLE employees ENABLE ROW LEVEL SECURITY;
ALTER TABLE reports ENABLE ROW LEVEL SECURITY;

DROP POLICY IF EXISTS "Public Access Employees" ON employees;
CREATE POLICY "Public Access Employees" ON employees FOR ALL USING (true) WITH CHECK (true);

DROP POLICY IF EXISTS "Public Access Reports" ON reports;
CREATE POLICY "Public Access Reports" ON reports FOR ALL USING (true) WITH CHECK (true);
